# The forms this round mends, one row each, read by test/wall-prefixes.test.mjs. Each is asked of wallHeld (the
# function the hook asks); nothing here is ever run, and git is a stand-in function that answers the wall's questions.
#
# A row is:  verdict | family | text
#   verdict  held <why>  wallHeld holds the line, and its first act's reason id is <why>
#            not held    wallHeld answers null (an everyday line that must still go through)
#   family   A leading redirects and assignments in any order; B a variable as the program; C gh/glab api graphql;
#            D a deploy script behind interpreter options; E an interpreter's release-notes script
#   text     the command line; a line break inside it is written as the two characters \n
# The session stands in its own folder /w/shop on the work branch feature of a repository whose default branch is main.
# Lines that begin with # and empty lines are not rows.

# ---- A: leading redirection targets and assignments, in any order and number, do not hide the program ----
held admin | A | x=1 2> err.txt sudo su
held admin | A | 2> err.txt x=1 sudo su
held admin | A | x=1 > a 2> b y=2 sudo su
held admin | A | FOO=a >e.txt sudo su
held admin | A | FOO=a > e.txt sudo su
held admin | A | > e.txt FOO=a sudo su
held admin | A | 2> e.txt FOO=a sudo su
held admin | A | x=1 2>e.txt > out.txt sudo su
not held | A | npm test 2> err.txt
not held | A | 2> err.txt npm test
not held | A | x=1 2> err.txt npm test
not held | A | FOO=a npm test
not held | A | npm test > "$LOG"
not held | A | df -h > "$OUT"
# an append (NAME+=…) or array-element (NAME[i]=…) assignment before the program is a leading assignment too
held admin | A | x+=1 sudo su
held admin | A | a[0]=1 sudo su
held admin | A | x+=1 2> e.txt sudo su
held admin | A | a[0]=1 2> e.txt sudo su
held admin | A | x+=1 > e.txt sudo su
held release | A | FOO+=bar npm publish
not held | A | CI+=1 npm test

held unreadable | B | > err.txt "$EDITOR"

# a bare variable glued to quotes or to text, where the text after the last variable has no path separator
held unreadable | B | ''$PROG
